cmake_minimum_required(VERSION 3.16)
project(random_art_generator LANGUAGES CXX)

find_package(Threads REQUIRED)
find_package(JPEG QUIET)

add_executable(random_art_generator src/main.cpp)
target_compile_features(random_art_generator PRIVATE cxx_std_17)
target_link_libraries(random_art_generator PRIVATE Threads::Threads)

if(WIN32)
    add_executable(random_art_gui WIN32 src/main.cpp)
    target_compile_definitions(random_art_gui PRIVATE RANDOM_ART_GUI=1)
    target_link_libraries(random_art_gui PRIVATE Threads::Threads user32 gdi32 comdlg32)
else()
    find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
    qt_add_executable(random_art_gui src/main.cpp)
    target_compile_definitions(random_art_gui PRIVATE RANDOM_ART_GUI=1)
    target_link_libraries(random_art_gui PRIVATE Threads::Threads Qt6::Widgets Qt6::Concurrent)
endif()
target_compile_features(random_art_gui PRIVATE cxx_std_17)

if(WIN32)
    target_link_libraries(random_art_generator PRIVATE user32)
elseif(APPLE)
    find_library(COREGRAPHICS_FRAMEWORK CoreGraphics)
    if(NOT COREGRAPHICS_FRAMEWORK)
        message(FATAL_ERROR "CoreGraphics is required to detect the native display resolution.")
    endif()
    target_link_libraries(random_art_generator PRIVATE ${COREGRAPHICS_FRAMEWORK})
elseif(UNIX)
    find_package(X11 QUIET)
    if(X11_FOUND)
        target_compile_definitions(random_art_generator PRIVATE RANDOM_ART_HAS_X11=1)
        target_include_directories(random_art_generator PRIVATE ${X11_INCLUDE_DIR})
        target_link_libraries(random_art_generator PRIVATE ${X11_LIBRARIES})
    endif()
endif()

if(JPEG_FOUND)
    target_compile_definitions(random_art_generator PRIVATE RANDOM_ART_HAS_JPEG=1)
    target_link_libraries(random_art_generator PRIVATE JPEG::JPEG)
    target_compile_definitions(random_art_gui PRIVATE RANDOM_ART_HAS_JPEG=1)
    target_link_libraries(random_art_gui PRIVATE JPEG::JPEG)
endif()

if(MSVC)
    target_compile_options(random_art_generator PRIVATE /W4 /permissive-)
    target_compile_options(random_art_gui PRIVATE /W4 /permissive-)
else()
    target_compile_options(random_art_generator PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(random_art_gui PRIVATE -Wall -Wextra -Wpedantic)
endif()
